% =============================================================================
%  main.tex
%  -----------------------------------------------------------------------------
%  A LaTeX template for academic manuscripts.
%  English-first, with full Chinese (CJK) support via XeLaTeX + xeCJK.
%  英文为主的学术论文 LaTeX 模板，内置中文支持（XeLaTeX + xeCJK）。
%
%  COMPILE (编译方式):
%     xelatex main.tex
%     bibtex  main        (if you use \cite / \citep / \citet)
%     xelatex main.tex
%     xelatex main.tex
%
%  Folders (目录结构):
%     text/    -- section files (正文分节文件)
%     tables/  -- table files    (表格文件)
%     figures/ -- figure files   (每图一个子目录：TikZ 源 + 编译好的 PDF)
% =============================================================================

\documentclass[12pt]{article}

% ---- Fonts & CJK support (must be loaded before inputting settings) ----
\usepackage{fontspec}   % XeLaTeX font management
\usepackage{xeCJK}      % Chinese support

% ---- General packages used by the body text ----
\usepackage{graphicx}     % figures
\usepackage{subcaption}   % subfigures
\usepackage{longtable}
\usepackage{array}
\usepackage{makecell}
\usepackage{ragged2e}

% ---- Global preamble settings (fonts, margins, line spacing, tables, math,
%      citations, hyperlinks) ----
\input{A_PreambleSettings.tex}

% ---- Paragraph indentation: 2em (standard for Chinese manuscripts) ----
\setlength{\parindent}{2em}

% =============================================================================
%  TITLE PAGE  --  REPLACE with your own title and author information.
%  请替换为你自己的论文标题与作者信息。
% =============================================================================
\title{\textbf{Your Manuscript Title:\\ A Reproducible LaTeX Template}}

\author{
  Your Name\thanks{Department, University. Email: your.name@university.edu}
  \and
  Co-author Name\thanks{Corresponding author. Department, University. Email: coauthor@university.edu}
}

\date{\today}

\begin{document}

\maketitle

% =============================================================================
%  ABSTRACT  --  bilingual example. Keep the English version as the main text;
%  the Chinese version below is optional and can be removed for English-only
%  submissions. 摘要示例：英文为主，中文可选。
% =============================================================================
\begin{abstract}
\input{text/abstract}
\end{abstract}

\newpage

% =============================================================================
%  MAIN TEXT  --  the structure below is a generic skeleton. Each section is
%  split into per-section files under text/. Rename the sections to match
%  your own paper.
%  正文结构：通用论文骨架，各节拆分为 text/ 下的独立文件，请按需改名。
% =============================================================================

\section{INTRODUCTION}
\input{text/introduction}

\section{THE ARGUMENT}
\input{text/THE ARGUMENT/intro}

\subsection{First Component of the Argument}
\label{sec:argument-part-1}
\input{text/THE ARGUMENT/part-1}

\subsection{Second Component of the Argument}
\label{sec:argument-part-2}
\input{text/THE ARGUMENT/part-2}

\subsection{Third Component of the Argument}
\label{sec:argument-part-3}
\input{text/THE ARGUMENT/part-3}

\section{HISTORICAL BACKGROUND}
\input{text/HISTORICAL BACKGROUND/intro}

\subsection{Background}
\label{sec:background}
\input{text/HISTORICAL BACKGROUND/background}

\subsection{Developments over Time}
\label{sec:developments}
\input{text/HISTORICAL BACKGROUND/developments}

\section{RESEARCH DESIGN}
\input{text/RESEARCH DESIGN/intro}

\subsection{Sample}
\label{sec:sample}
\input{text/RESEARCH DESIGN/sample}

\subsection{Key Variable}
\label{sec:key-variable}
\input{text/RESEARCH DESIGN/key-variable}

\subsection{Outcome 1}
\label{sec:outcome-1}
\input{text/RESEARCH DESIGN/outcome-1}

\subsection{Outcome 2}
\label{sec:outcome-2}
\input{text/RESEARCH DESIGN/outcome-2}

\section{RESULTS}
\input{text/RESULTS/intro}

\subsection{Main Results}
\label{sec:main-results}
\input{text/RESULTS/main-results}

\subsection{Robustness Checks}
\label{sec:robustness}
\input{text/RESULTS/robustness}

\section{CONCLUSION}
\label{sec:conclusion}
\input{text/conclusion}

\newpage

% =============================================================================
%  REFERENCES  --  aeaown.bst is a Harvard (author-year) style.
% =============================================================================
\bibliographystyle{aeaown}
\bibliography{references}

\newpage

% =============================================================================
%  APPENDIX  --  table and figure numbering restarts with B/A prefixes.
% =============================================================================
\appendix

\section*{APPENDIX}
\addcontentsline{toc}{section}{APPENDIX}

\section{Tables}
\label{app:tables}
\setcounter{table}{0}
\renewcommand{\thetable}{B\arabic{table}}
\input{text/APPENDIX/tables.tex}

\section{Figures}
\label{app:figures}
\setcounter{figure}{0}
\renewcommand{\thefigure}{A\arabic{figure}}
\input{text/APPENDIX/figures.tex}

\end{document}
